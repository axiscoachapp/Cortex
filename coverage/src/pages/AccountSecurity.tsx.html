import {useState} from 'react';
import {Link} from 'react-router-dom';
import {useQuery,useQueryClient} from '@tanstack/react-query';
import {supabase} from '@/integrations/supabase/client';
import {useAuth} from '@/contexts/AuthContext';
import {Button} from '@/components/ui/button';
import {Input} from '@/components/ui/input';
import {BrandLockup} from '@/components/BrandLockup';
export default function AccountSecurity(){
  const {user}=useAuth();const client=useQueryClient();const [setup,setSetup]=useState<{id:string;qr:string;secret:string}|null>(null);const [code,setCode]=useState('');const [factor,setFactor]=useState('');const [error,setError]=useState('');const [notice,setNotice]=useState('');const [busy,setBusy]=useState(false);
  const factors=useQuery({queryKey:['mfa-factors',user?.id],queryFn:async()=>{const {data,error}=await supabase.auth.mfa.listFactors();if(error)throw error;return data.totp;}});
  const run=async(action:()=>Promise<void>)=>{setBusy(true);setError('');setNotice('');try{await action();}catch{setError('Não foi possível concluir. Confira o código e a conexão e tente novamente.');}finally{setBusy(false);}};
  const enroll=()=>run(async()=>{const {data,error}=await supabase.auth.mfa.enroll({factorType:'totp',friendlyName:'Cortex '+new Date().toISOString().slice(0,10),issuer:'Haddad Nicola'});if(error)throw error;setSetup({id:data.id,qr:data.totp.qr_code,secret:data.totp.secret});setFactor(data.id);});
  const verify=()=>run(async()=>{const {error}=await supabase.auth.mfa.challengeAndVerify({factorId:setup?.id||factor,code});if(error)throw error;setSetup(null);setCode('');setNotice('Verificação concluída. Sua sessão está protegida pelo segundo fator.');void client.invalidateQueries({queryKey:['mfa-factors']});void client.invalidateQueries({queryKey:['session-assurance']});});
  const remove=(id:string)=>run(async()=>{if(!window.confirm('Remover este autenticador da sua conta?'))return;const {error}=await supabase.auth.mfa.unenroll({factorId:id});if(error)throw error;setSetup(null);void client.invalidateQueries({queryKey:['mfa-factors']});setNotice('Autenticador removido.');});
  return <main className="min-h-screen max-w-2xl mx-auto p-6 space-y-6"><Link to="/" className="text-sm underline">Voltar ao início</Link><BrandLockup/><h1 className="font-serif text-3xl">Segurança da conta</h1><p className="text-sm text-muted-foreground">Use um aplicativo autenticador para verificar o acesso com um código temporário. Cada médico deve manter sua própria conta.</p>
  {error&&<p role="alert" className="text-destructive text-sm">{error}</p>}{notice&&<p role="status" className="text-sm">{notice}</p>}
  {factors.isError?<p role="alert">Não foi possível carregar autenticadores. <button onClick={()=>factors.refetch()} className="underline">Tentar novamente</button></p>:factors.isLoading?<p role="status">Carregando…</p>:<ul className="space-y-3">{factors.data?.map(f=><li key={f.id} className="p-4 border rounded-lg flex justify-between items-center gap-3"><span className="text-sm">{f.friendly_name||'Aplicativo autenticador'} · {f.status==='verified'?'Ativo':'Configuração pendente'}</span><Button variant="outline" size="sm" disabled={busy} onClick={()=>remove(f.id)}>Remover</Button></li>)}</ul>}
  {!setup&&<Button disabled={busy||factors.isLoading||factors.isError} onClick={enroll}>Adicionar autenticador</Button>}
  {setup&&<section className="rounded-xl border bg-card p-5 space-y-4"><h2 className="font-semibold">Configurar aplicativo</h2><p className="text-sm">Leia o QR code no seu aplicativo. Guarde sua configuração em um local seguro e confirme o código abaixo.</p><img src={setup.qr.startsWith('data:')?setup.qr:'data:image/svg+xml;charset=utf-8,'+encodeURIComponent(setup.qr)} alt="QR code para configurar seu autenticador" className="w-52 h-52"/><details><summary className="cursor-pointer text-sm">Inserir chave manualmente</summary><code className="block break-all py-3 select-all">{setup.secret}</code></details><Button variant="outline" disabled={busy} onClick={()=>remove(setup.id)}>Cancelar configuração</Button></section>}
  {(setup||factors.data?.some(f=>f.status==='verified'))&&<form className="space-y-3" onSubmit={e=>{e.preventDefault();void verify();}}>{!setup&&<label className="block text-sm">Autenticador<select aria-label="Autenticador" className="block w-full border rounded p-2 mt-2" value={factor} onChange={e=>setFactor(e.target.value)} required><option value="">Selecione</option>{factors.data?.filter(f=>f.status==='verified').map(f=><option key={f.id} value={f.id}>{f.friendly_name||f.id}</option>)}</select></label>}<label className="block text-sm space-y-2"><span>Código do aplicativo</span><Input autoComplete="one-time-code" inputMode="numeric" pattern="[0-9]{6}" maxLength={6} value={code} onChange={e=>setCode(e.target.value.replace(/\D/g,''))} required/></label><Button type="submit" disabled={busy||code.length!==6||(!setup&&!factor)}>Verificar código</Button></form>}
  <p className="text-xs text-muted-foreground border-t pt-4">Se perder acesso ao autenticador, solicite recuperação ao responsável pelo ambiente. Desabilitar um fator exige uma sessão verificada.</p>
  </main>;
}
