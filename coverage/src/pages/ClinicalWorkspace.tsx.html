type Item = Tables<'clinical_work_items'>;
const statusLabels: Record<string,string> = {open:'Pendente',draft:'Rascunho',done:'Concluído',approved:'Aprovado',cancelled:'Encerrado'};
function detailsOf(item?:Item):Record<string,string> {
  return item?.details && typeof item.details==='object' && !Array.isArray(item.details)
    ? Object.fromEntries(Object.entries(item.details).filter((entry):entry is [string,string]=>typeof entry[1]==='string')) : {};
}
 
function workSnapshotText(snapshot:Json):string {
 if(!snapshot||typeof snapshot!=='object'||Array.isArray(snapshot))return '';
 const detail=snapshot.details&&typeof snapshot.details==='object'&&!Array.isArray(snapshot.details)?snapshot.details:{};
 const state=typeof detail.state==='string'?({active:'Ativo',changed:'Alterado',stopped:'Suspenso'} as Record<string,string>)[detail.state]:'';
 return [snapshot.title,snapshot.content,typeof detail.date==='string'?'Data: '+formatDateOnly(detail.date):'',detail.value?'Valor: '+detail.value+' '+(detail.unit??''):'',state?'Estado: '+state:'',detail.source?'Fonte: '+detail.source:''].filter(Boolean).join('\n\n');
}
